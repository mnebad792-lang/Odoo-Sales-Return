import streamlit as st import pandas as pd import requests from io import BytesIO # إعداد الصفحة لتكون بعرض واسع (Wide Mode) لاستيعاب الجدول بشكل مريح st.set_page_config(page_title="نظام تحليل ومقارنة الفرق والمناديب", layout="wide") # تخصيص التصميم والتحكم في عرض أعمدة الجدول عبر CSS لتجنب أي تداخل في التواريخ والأرقام st.markdown(""" """, unsafe_allow_html=True) st.markdown("

نظام تحليل ومقارنة الفرق والمناديب (مبيعات، مرتجعات، صافي والاتجاهات)

", unsafe_allow_html=True) # رابط ملف Google Sheets GOOGLE_SHEET_CSV_URL = 'https://docs.google.com/spreadsheets/d/1SVnlPRdw9bNqaA76RM7qa4advkC_nybW/export?format=csv&gid=1841364794' @st.cache_data(show_spinner=True) def load_data(file=None): try: if file is not None: if file.name.endswith('.csv'): df = pd.read_csv(file) else: df = pd.read_excel(file) else: # محاولة الجلب السحابي response = requests.get(GOOGLE_SHEET_CSV_URL) if response.status_code == 200: df = pd.read_csv(BytesIO(response.content)) else: st.error("فشل الجلب السحابي، يرجى رفع الملف يدوياً.") return pd.DataFrame() return df except Exception as e: st.error(f"حدث خطأ أثناء تحميل الملف: {e}") return pd.DataFrame() # قسم الرفع اليدوي أو الجلب col_up1, col_up2 = st.columns([3, 1]) with col_up1: uploaded_file = st.file_uploader("📁 ارفع ملف الإكسيل أو CSV هنا:", type=["xlsx", "xls", "csv"]) df = load_data(uploaded_file) if not df.empty: st.success(f"✅ تم تحميل البيانات بنجاح. عدد السجلات الإجمالي: {len(df):,}") # محاولة التعرف على الأعمدة لتطبيق التنسيقات الذكية cols = df.columns.tolist() # فلاتر البحث السريع st.markdown("---") st.subheader("🔍 محرك الفلاتر الشامل:") c1, c2, c3, c4 = st.columns(4) with c1: client_search = st.text_input("ابحث عن العميل:") with c2: rep_search = st.text_input("ابحث عن المندوب:") with c3: team_search = st.text_input("ابحث عن الفريق:") with c4: # البحث عن عمود السنة إن وجد date_col = next((c for c in cols if 'تاريخ' in str(c) or 'date' in str(c).lower()), None) years = [] if date_col: df[date_col] = pd.to_datetime(df[date_col], errors='coerce') years = sorted(df[date_col].dt.year.dropna().unique().astype(int).tolist()) selected_year = st.selectbox("السنة:", ["كل السنوات"] + years) # تطبيق الفلاتر df_filtered = df.copy() if client_search: client_col = next((c for c in cols if 'عميل' in str(c) or 'customer' in str(c).lower()), cols[2] if len(cols)>2 else cols[0]) df_filtered = df_filtered[df_filtered[client_col].astype(str).str.contains(client_search, case=False, na=False)] if rep_search: rep_col = next((c for c in cols if 'مندوب' in str(c) or 'rep' in str(c).lower()), cols[1] if len(cols)>1 else cols[0]) df_filtered = df_filtered[df_filtered[rep_col].astype(str).str.contains(rep_search, case=False, na=False)] if team_search: team_col = next((c for c in cols if 'فريق' in str(c) or 'team' in str(c).lower()), cols[0]) df_filtered = df_filtered[df_filtered[team_col].astype(str).str.contains(team_search, case=False, na=False)] if selected_year != "كل السنوات" and date_col: df_filtered = df_filtered[df_filtered[date_col].dt.year == int(selected_year)] st.markdown(f"**عدد السجلات المعروضة بعد التصفية:** {len(df_filtered):,}") # ضبط تكوين الأعمدة وتوسيع المساحات للأرقام والتواريخ وتصغير النصوص الطويلة لتجنب التداخل column_config = {} for col in cols: col_lower = str(col).lower() if 'فريق' in col_lower or 'team' in col_lower: column_config[col] = st.column_config.TextColumn(col, width="small") elif 'مندوب' in col_lower or 'rep' in col_lower: column_config[col] = st.column_config.TextColumn(col, width="small") elif 'عميل' in col_lower or 'customer' in col_lower: column_config[col] = st.column_config.TextColumn(col, width="medium") elif 'منتج' in col_lower or 'item' in col_lower or 'product' in col_lower: column_config[col] = st.column_config.TextColumn(col, width="medium") elif 'تاريخ' in col_lower or 'date' in col_lower: column_config[col] = st.column_config.DateColumn(col, width="small", format="YYYY-MM-DD") elif 'رقم' in col_lower or 'invoice' in col_lower or 'order' in col_lower: column_config[col] = st.column_config.TextColumn(col, width="small") elif any(w in col_lower for w in ['مبلغ', 'قيمة', 'إجمالي', 'سعر', 'كمية', 'خصم']): column_config[col] = st.column_config.NumberColumn(col, format="%.2f", width="small") # عرض الجدول مع التنسيق المحسّن st.dataframe( df_filtered, use_container_width=True, column_config=column_config, height=500 ) else: st.info("الرجاء رفع ملف البيانات أو التأكد من توفرها لبدء العرض.")
